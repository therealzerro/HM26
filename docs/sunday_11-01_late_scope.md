```sql
-- §1 nine sets over the 26-state late pool
with late as (select unnest(array['AZ','CA','CO','CT','FL','IA','ID','IL','IN','KS','KY','LA','MO','MS','NC','NE','NJ','NM','NY','OK','ON','QC','VA','W.Canada','WA','WI']) j),
cand as (select * from (values ('{2,8,9}','928'),('{0,2,3}','302'),('{0,1,4}','401'),('{0,4,9}','904'),('{4,5,7}','574'),('{0,4,6}','046'),('{1,5,7}','571'),('{0,5,7}','750'),('{2,6,8}','826')) v(cs, posted)),
d as (select h.* from histories h join late on late.j=h.jurisdiction where h.session='evening' and h.date_et between '2026-04-01' and '2026-10-31'),
tot as (select count(*) n_all, count(*) filter (where date_et >= '2026-09-01') n60, count(*) filter (where date_et >= '2026-10-01') n30, count(*) filter (where extract(dow from date_et)=0) n_sun from d)
select c.cs, c.posted,
 count(d.id) filter (where d.comboset_sorted=c.cs) box_all, round((select n_all from tot)*6/1000.0,1) exp_all,
 count(d.id) filter (where d.comboset_sorted=c.cs and d.date_et>='2026-09-01') box60, round((select n60 from tot)*6/1000.0,1) exp60,
 count(d.id) filter (where d.comboset_sorted=c.cs and d.date_et>='2026-10-01') box30, round((select n30 from tot)*6/1000.0,1) exp30,
 count(d.id) filter (where d.comboset_sorted=c.cs and extract(dow from d.date_et)=0) box_sun, round((select n_sun from tot)*6/1000.0,1) exp_sun,
 count(d.id) filter (where d.result_digits=c.posted) exact_all, round((select n_all from tot)/1000.0,2) exp_exact,
 count(d.id) filter (where d.result_digits=c.posted and d.date_et>='2026-09-01') exact60,
 count(distinct d.jurisdiction) filter (where d.comboset_sorted=c.cs) states,
 max(d.date_et) filter (where d.comboset_sorted=c.cs) last_late
from cand c left join d on true group by c.cs, c.posted order by box30 desc, box60 desc;

-- §2 orders
select comboset_sorted, result_digits, count(*), count(*) filter (where date_et>='2026-09-01') n60, count(*) filter (where date_et>='2026-10-01') n30, max(date_et)
from <late rows d> where comboset_sorted in ('{2,8,9}','{1,5,7}') group by 1,2 order by 1, 3 desc;

-- §3 per state
with per as (select jurisdiction, count(*) draws from <late rows d> group by 1)
select v.cs, s.jurisdiction, p.draws, round(p.draws*6/1000.0,2) exp_box, count(d.id) box,
 count(d.id) filter (where d.date_et>='2026-09-01') box60,
 count(d.id) filter (where d.result_digits = case v.cs when '{2,8,9}' then '928' else '571' end) exact_posted,
 string_agg(d.result_digits||' '||to_char(d.date_et,'MM-DD'), ', ' order by d.date_et) draws_list,
 (select count(*) from <late rows> x where x.jurisdiction=s.jurisdiction and x.date_et > max(d.date_et)) draws_since
from (values ('{2,8,9}'),('{1,5,7}')) v(cs) cross join per s
left join <late rows> d on d.jurisdiction=s.jurisdiction and d.comboset_sorted=v.cs join per p on p.jurisdiction=s.jurisdiction
group by v.cs, s.jurisdiction, p.draws having count(d.id)>0 order by v.cs, box desc, box60 desc, draws_since;

-- §4 pair repeats (three unordered digit pairs per draw)
with d as (select date_et, session, jurisdiction, result_digits from histories where jurisdiction is not null and ((date_et='2026-11-01' and session='midday') or (date_et='2026-10-31' and session='evening'))),
p as (select date_et, session, jurisdiction, result_digits, least(substr(result_digits,1,1),substr(result_digits,2,1))||greatest(substr(result_digits,1,1),substr(result_digits,2,1)) pr from d
 union all select date_et, session, jurisdiction, result_digits, least(substr(result_digits,1,1),substr(result_digits,3,1))||greatest(substr(result_digits,1,1),substr(result_digits,3,1)) from d
 union all select date_et, session, jurisdiction, result_digits, least(substr(result_digits,2,1),substr(result_digits,3,1))||greatest(substr(result_digits,2,1),substr(result_digits,3,1)) from d)
select date_et, session, pr, count(*) n, string_agg(jurisdiction||' '||result_digits, ' ' order by jurisdiction) from p group by 1,2,3 having count(*)>=3 order by 1 desc, 4 desc;

-- midday 11/01 closures check
select jurisdiction, result_digits, comboset_sorted from histories where date_et='2026-11-01' and session='midday' and jurisdiction is not null order by 1;
```
