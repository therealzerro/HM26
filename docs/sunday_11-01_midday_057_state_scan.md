## SQL (re-runnable in the Supabase SQL editor)

```sql
-- §1 per-state lenses for {0,5,7}, midday, Sunday-live, 2026-04-01 → 2026-10-31
with d as (
  select jurisdiction, date_et, result_digits, comboset_sorted,
         (substr(comboset_sorted,2,1)<>substr(comboset_sorted,4,1) and substr(comboset_sorted,4,1)<>substr(comboset_sorted,6,1)) single,
         (substr(result_digits,1,1)::int+substr(result_digits,2,1)::int+substr(result_digits,3,1)::int) dsum,
         ((position('0' in result_digits)>0)::int+(position('5' in result_digits)>0)::int+(position('7' in result_digits)>0)::int) nset,
         (position('5' in result_digits)>0 and position('7' in result_digits)>0) pair57,
         (position('0' in result_digits)>0 and position('5' in result_digits)>0) pair05,
         (position('0' in result_digits)>0 and position('7' in result_digits)>0) pair07,
         (substr(result_digits,1,2)='05' or substr(result_digits,2,2)='57') pos057,
         (substr(result_digits,1,2)='75' or substr(result_digits,2,2)='50') pos750
  from histories where session='midday' and date_et between '2026-04-01' and '2026-10-31'
    and jurisdiction not in ('TN','SC','AR','TX','WV','PR')),
rn as (select d.*, row_number() over (partition by jurisdiction order by date_et desc, result_digits) rn from d)
select jurisdiction, count(*) draws,
  sum((comboset_sorted='{0,5,7}')::int) box, round(count(*)*6/1000.0,2) exp_box,
  sum((result_digits='057')::int) s057, sum((result_digits='750')::int) s750,
  string_agg(result_digits||' '||to_char(date_et,'MM-DD'), ', ' order by date_et) filter (where comboset_sorted='{0,5,7}') draws_of_set,
  min(rn) filter (where comboset_sorted='{0,5,7}')-1 since,
  sum((nset>=2)::int) two_of, round(count(*)*150/1000.0,1) exp_two,
  sum((nset>=2 and rn<=30)::int) two_of_l30, sum((nset>=2 and rn<=60)::int) two_of_l60,
  sum(pair57::int) p57, sum(pair05::int) p05, sum(pair07::int) p07, round(count(*)*54/1000.0,1) exp_pair,
  sum(pos057::int) pos_057, sum(pos750::int) pos_750, round(count(*)*19/1000.0,1) exp_pos,
  sum((dsum=12)::int) sum12, round(count(*)*73/1000.0,1) exp_sum12,
  round(100.0*sum(single::int)/count(*),1) pct_singles
from rn group by jurisdiction order by sum((comboset_sorted='{0,5,7}')::int)-count(*)*6/1000.0 desc, two_of_l30 desc;

-- §2 last 12 midday draws
select jurisdiction, string_agg(result_digits, ' ' order by date_et desc, result_digits) last12 from (
  select jurisdiction, date_et, result_digits, row_number() over (partition by jurisdiction order by date_et desc, result_digits) rn
  from histories where session='midday' and date_et<='2026-10-31' and jurisdiction in ('KY','NC','NY','AZ','ME,NH,VT','IA','MO','PA','IL')) x
where rn<=12 group by jurisdiction order by jurisdiction;

-- engine lens: {0,5,7} on past Midday boards and the midday draws of it those days
select slate_date, posted_order from adaptive_tracking
 where scope='midday' and combo_set='{0,5,7}' and slate_date between '2026-04-01' and '2026-10-31' group by 1,2 order by 1;
select h.date_et, h.jurisdiction, h.result_digits from histories h
 where h.session='midday' and h.comboset_sorted='{0,5,7}'
   and h.date_et in (select slate_date from adaptive_tracking where scope='midday' and combo_set='{0,5,7}');
```
