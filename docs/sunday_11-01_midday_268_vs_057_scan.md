## SQL (re-runnable in the Supabase SQL editor)

```sql
-- §1 pool windows for both sets
with d as (
  select jurisdiction, date_et, result_digits, comboset_sorted,
         (jurisdiction not in ('TN','SC','AR','TX','WV','PR')) sun_live,
         extract(dow from date_et)=0 is_sun,
         ((position('2' in result_digits)>0)::int+(position('6' in result_digits)>0)::int+(position('8' in result_digits)>0)::int) n268,
         ((position('0' in result_digits)>0)::int+(position('5' in result_digits)>0)::int+(position('7' in result_digits)>0)::int) n057
  from histories where session='midday' and date_et between '2026-04-01' and '2026-10-31')
select s.set_name, w.win, count(*) draws, round(count(*)*6/1000.0,2) exp_box,
  sum((comboset_sorted=s.cs)::int) box,
  sum((result_digits=s.op)::int) str_op, sum((result_digits=s.po)::int) str_posted, round(count(*)/1000.0,2) exp_str,
  sum((case when s.cs='{2,6,8}' then n268 else n057 end>=2)::int) two_of, round(count(*)*150/1000.0,1) exp_two,
  count(distinct jurisdiction) filter (where comboset_sorted=s.cs) states_drew
from d
cross join (values ('{2,6,8}','268','826','268 / set {2,6,8}'),('{0,5,7}','057','750','057 / set {0,5,7}')) s(cs,op,po,set_name)
cross join (values ('all since 4/01',1),('Sunday-live states since 4/01',2),('last 60d (9/02→)',3),('last 30d (10/02→)',4),('Sundays only',5)) w(win,k)
where (w.k=1) or (w.k=2 and sun_live) or (w.k=3 and date_et>='2026-09-02') or (w.k=4 and date_et>='2026-10-02') or (w.k=5 and is_sun)
group by 1,2,w.k order by 1 desc, w.k;

-- §2/§3 per state (swap the set literal, digits and orders for {0,5,7} / '057','750')
with d as (
  select jurisdiction, date_et, result_digits, comboset_sorted,
         ((position('2' in result_digits)>0)::int+(position('6' in result_digits)>0)::int+(position('8' in result_digits)>0)::int) nset
  from histories where session='midday' and date_et between '2026-04-01' and '2026-10-31'),
rn as (select d.*, row_number() over (partition by jurisdiction order by date_et desc, result_digits) rn from d)
select jurisdiction, count(*) draws,
  sum((comboset_sorted='{2,6,8}')::int) box, round(count(*)*6/1000.0,2) exp_box,
  sum((result_digits in ('268','826'))::int) str,
  string_agg(result_digits||' '||to_char(date_et,'MM-DD'), ', ' order by date_et) filter (where comboset_sorted='{2,6,8}') draws_of_set,
  min(rn) filter (where comboset_sorted='{2,6,8}')-1 draws_since,
  sum((nset>=2)::int) two_of, round(count(*)*150/1000.0,1) exp_two,
  sum((nset>=2 and rn<=30)::int) two_of_last30,
  sum((comboset_sorted='{2,6,8}' and date_et>='2026-09-02')::int) box_60d, round(sum((date_et>='2026-09-02')::int)*6/1000.0,2) exp_box_60d,
  (jurisdiction not in ('TN','SC','AR','TX','WV','PR')) sunday_live
from rn group by jurisdiction order by box-exp_box desc, two_of_last30 desc;

-- board check
select scope, combo_set, posted_order from adaptive_tracking
 where slate_date='2026-11-01' and combo_set in ('{2,6,8}','{0,5,7}') order by 1,2;
```
